# v0.6 One-Hour Meeting Proof

Run: `2026-05-14-v0.6-summary-signed-records`
Date: 2026-05-14

## Scope

This proof uses deterministic local fixtures, as allowed by the v0.6
manifest when real TSA or Ollama services are unavailable. It verifies the
release-plan exit criterion that a one-hour meeting can produce an
operator-approved sourced summary and a signed-record deterministic contract
fixture, not a valid PDF/A-3B document in v1.0.0.

## Command

```powershell
@'
from datetime import UTC, datetime
from civiccast.captions import CaptionCue
from civiccast.summary.generate import SummaryGenerationPipeline
from civiccast.summary.models import OperatorApproval
from civiccast.summary.store import InMemorySummaryStore
from civiccast.records.exporter import SignedRecordExporter
from civiccast.records.pdfa import validate_pdfa3_contract
from civiccast.records.timestamp import verify_timestamp_proof

cues = []
for idx in range(60):
    start = float(idx * 60)
    end = start + 45.0
    if idx == 12:
        text = "Council voted 5-2 to approve the water main contract for $250,000."
    elif idx == 27:
        text = "Member Rivera moved to amend the budget and Member Chen seconded the motion."
    elif idx == 44:
        text = "The roll call was 4 ayes and 1 nay for the amended permit schedule."
    else:
        text = f"Public meeting discussion segment {idx + 1} covered agenda item {idx % 6 + 1}."
    cues.append(CaptionCue(cue_id=f"cue-{idx:03d}", start_seconds=start, end_seconds=end, text=text, confidence=0.96))

summary_store = InMemorySummaryStore()
summary = SummaryGenerationPipeline().generate(meeting_id="one-hour-v0.6-proof", cues=cues)
summary_store.create_summary(summary)
approval = OperatorApproval(summary_id=summary.summary_id, operator_id="operator-proof", operator_display_name="CivicCast Proof Operator", approved_at=datetime.now(UTC), approval_note="Deterministic one-hour v0.6 proof approval.")
approved = summary_store.approve_summary(approval)
record = SignedRecordExporter(summary_store=summary_store).export(summary_id=approved.summary_id, requested_by="operator-proof")
proof = validate_pdfa3_contract(record.pdf_bytes)
assert approved.status == "approved"
assert len(cues) == 60
assert cues[-1].end_seconds >= 3585.0
assert any(claim.claim_type == "quantitative" and claim.transcript_ranges for claim in approved.sourced_claims)
assert all(line.startswith("PASS") for line in proof)
assert verify_timestamp_proof(record.pdf_bytes, record.timestamp_proof)
print("ONE_HOUR_PROOF: PASS")
print(f"meeting_id={approved.meeting_id}")
print(f"cue_count={len(cues)}")
print(f"duration_seconds={cues[-1].end_seconds:.0f}")
print(f"summary_id={approved.summary_id}")
print(f"summary_status={approved.status}")
print(f"quantitative_claims={sum(1 for c in approved.sourced_claims if c.claim_type == 'quantitative')}")
print(f"record_id={record.record_id}")
print(f"artifact_digest={record.artifact_digest}")
for line in proof:
    print(line)
'@ | .\.venv\Scripts\python.exe -
```
