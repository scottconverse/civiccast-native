% CivicCast User Manual print styling.
%
% Design intent: this is an operator handbook, not a paper. It should look
% approachable to a clerk or a volunteer station operator reading it at a
% desk. Brand colours mirror the public site (docs/index.html): warm paper,
% navy accent, muted slate for secondary text.
%
% Dependency note: every package used here ships in the TeX packages the
% ci-docs workflow already installs (texlive-latex-recommended and
% texlive-latex-extra). Do not add a package outside that set without also
% updating .github/workflows/ci-docs.yml and ADR 0005.

\usepackage{xcolor}
\definecolor{ccink}{HTML}{101828}
\definecolor{ccaccent}{HTML}{2458A6}
\definecolor{ccdeep}{HTML}{16365C}
\definecolor{ccmuted}{HTML}{475467}
\definecolor{ccpaper}{HTML}{F7F2EA}
\definecolor{ccline}{HTML}{D8CDBF}
\definecolor{cccode}{HTML}{F4F6F8}

% -------------------------------------------------------------- cover page --
% Pandoc folds the subtitle into \title, so styling the title block styles
% both. Brand the cover instead of leaving the LaTeX default: sans, navy,
% and an accent rule, so the handbook does not open like a term paper.
\usepackage{titling}

\pretitle{%
  \vspace*{3.5cm}%
  \begin{center}%
  {\color{ccaccent}\rule{5cm}{3pt}}\par\vspace{1.2em}%
  \sffamily\Huge\bfseries\color{ccdeep}%
}
\posttitle{%
  \par\end{center}%
  \vspace{0.8em}%
  \begin{center}{\color{ccline}\rule{11cm}{0.8pt}}\end{center}%
}
\preauthor{\begin{center}\sffamily\large\color{ccmuted}}
\postauthor{\par\end{center}}
\predate{\begin{center}\sffamily\normalsize\color{ccmuted}}
\postdate{\par\end{center}}

% ---------------------------------------------------------------- headings --
\usepackage{titlesec}

\titleformat{\section}
  {\sffamily\LARGE\bfseries\color{ccdeep}}{\thesection}{0.6em}{}
  [\vspace{1pt}{\color{ccline}\titlerule[1pt]}]
\titleformat{\subsection}
  {\sffamily\large\bfseries\color{ccaccent}}{\thesubsection}{0.6em}{}
\titleformat{\subsubsection}
  {\sffamily\normalsize\bfseries\color{ccink}}{\thesubsubsection}{0.6em}{}

\titlespacing*{\section}{0pt}{2.6ex plus 1ex minus .2ex}{1.4ex plus .2ex}
\titlespacing*{\subsection}{0pt}{2.0ex plus .8ex minus .2ex}{0.9ex plus .2ex}
\titlespacing*{\subsubsection}{0pt}{1.6ex plus .6ex minus .2ex}{0.7ex plus .2ex}

% ------------------------------------------------------- callouts (quotes) --
% The manual uses "> " blockquotes for the things a reader must not miss
% (safety boundaries, "do not install this build", stop-and-report rules).
% Default LaTeX renders those as a limp indented paragraph. Give them a
% tinted panel with an accent bar so they read as callouts.
\usepackage[most]{tcolorbox}

\newtcolorbox{cccallout}{
  enhanced,
  breakable,
  colback=ccpaper,
  colframe=ccaccent,
  boxrule=0pt,
  leftrule=3pt,
  arc=1pt,
  outer arc=1pt,
  boxsep=0pt,
  left=11pt, right=11pt, top=9pt, bottom=9pt,
}
\renewenvironment{quote}{\begin{cccallout}}{\end{cccallout}}

% ------------------------------------------------------- running head/foot --
% \ccmanualversion is the single source of truth for the version shown in the
% running header. render_user_manual.py and scripts/render-user-manual.sh both
% \renewcommand it from docs/USER-MANUAL.md's own YAML `subtitle` field before
% invoking pandoc, so the header cannot drift from the title page the way a
% hard-coded version string here once did (TW-A). The \providecommand default
% only fires if this file is ever compiled without that override.
\providecommand{\ccmanualversion}{v0.0.0-unset}
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0pt}
\fancyhead[L]{\sffamily\footnotesize\color{ccmuted}CivicCast User Manual}
\fancyhead[R]{\sffamily\footnotesize\color{ccmuted}\ccmanualversion}
\fancyfoot[C]{\sffamily\footnotesize\color{ccmuted}\thepage}
\fancypagestyle{plain}{%
  \fancyhf{}%
  \renewcommand{\headrulewidth}{0pt}%
  \fancyfoot[C]{\sffamily\footnotesize\color{ccmuted}\thepage}%
}

% ------------------------------------------------------------- readability --
\usepackage{setspace}
\setstretch{1.12}
\setlength{\parskip}{0.55em}
\setlength{\emergencystretch}{3em}

% Softer rules in tables; pandoc emits booktabs.
\usepackage{booktabs}

% Lists that breathe a little.
\usepackage{enumitem}
\setlist[itemize]{itemsep=2pt, topsep=4pt, parsep=0pt}
\setlist[enumerate]{itemsep=2pt, topsep=4pt, parsep=0pt}

% Inline code and code blocks: light panel instead of bare monospace.
\usepackage{fvextra}
\fvset{breaklines=true, breakanywhere=true, fontsize=\small}
\definecolor{shadecolor}{HTML}{F4F6F8}
